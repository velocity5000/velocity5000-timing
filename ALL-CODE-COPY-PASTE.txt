function doGet(e) {
  const action = e && e.parameter ? String(e.parameter.action || "") : "";
  try {
    if (action === "config") return jsonOutput(getConfigData());
    if (action === "health") return jsonOutput({ success: true, application: "Velocity 5000 Timing System", version: "1.2.0", serverTime: new Date().toISOString() });
    if (action === "reviewData") return jsonOutput(getReviewData(String(e.parameter.eventId || "")));
    return jsonOutput({ success: true, application: "Velocity 5000 Timing System", message: "Backend is running.", availableActions: ["health", "config", "reviewData"] });
  } catch (err) {
    return jsonOutput({ success: false, error: err.message });
  }
}

function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) throw new Error("Missing request body.");
    const request = JSON.parse(e.postData.contents);
    let result;
    switch (request.action) {
      case "uploadRecords": result = uploadRecords(request); break;
      case "selectResult": requireTimer1_(request); result = selectResults_({ ...request, recordIds: [request.recordId] }); break;
      case "selectResults": requireTimer1_(request); result = selectResults_(request); break;
      case "excludeRecord": requireTimer1_(request); result = excludeRecords_({ ...request, recordIds: [request.recordId] }); break;
      case "excludeRecords": requireTimer1_(request); result = excludeRecords_(request); break;
      case "restoreRecord": requireTimer1_(request); result = restoreRecords_({ ...request, recordIds: [request.recordId] }); break;
      case "restoreRecords": requireTimer1_(request); result = restoreRecords_(request); break;
      case "correctBib": requireTimer1_(request); result = correctBib_(request); break;
      case "finalizeResults": requireTimer1_(request); result = finalizeResults_(request); break;
      default: throw new Error("Unknown POST action.");
    }
    return jsonOutput(result);
  } catch (err) {
    return jsonOutput({ success: false, error: err.message });
  }
}

/** Add missing required headers at the right. Never clear, reorder, or rename cells. */
function ensureHeaders_(sheet, requiredHeaders) {
  if (sheet.getLastRow() === 0 || sheet.getLastColumn() === 0) {
    sheet.getRange(1, 1, 1, requiredHeaders.length).setValues([requiredHeaders]);
    return sheet;
  }
  const lastCol = sheet.getLastColumn();
  const current = sheet.getRange(1, 1, 1, lastCol).getDisplayValues()[0].map(v => String(v || "").trim());
  const missing = requiredHeaders.filter(h => !current.includes(h));
  if (missing.length) sheet.getRange(1, lastCol + 1, 1, missing.length).setValues([missing]);
  return sheet;
}
function ensureSheet_(name, headers) {
  const ss = getSpreadsheet_();
  let sheet = ss.getSheetByName(name);
  if (!sheet) sheet = ss.insertSheet(name);
  return ensureHeaders_(sheet, headers);
}

/** Run manually once from the Apps Script editor; safe to rerun. */
function repairTimingSystemHeaders() {
  const ss = getSpreadsheet_();
  const required = {
    "Config": CONFIG_HEADERS,
    "Timer1": TIMER_HEADERS,
    "Timer2": TIMER_HEADERS,
    "Timer3": TIMER_HEADERS,
    "Timer4": TIMER_HEADERS,
    "Timer5": TIMER_HEADERS,
    "Consolidated": CONSOLIDATED_HEADERS,
    "CorrectionLog": CORRECTION_HEADERS
  };
  const report = [];
  Object.keys(required).forEach(name => {
    let sheet = ss.getSheetByName(name);
    if (!sheet) {
      sheet = ss.insertSheet(name);
      sheet.getRange(1, 1, 1, required[name].length).setValues([required[name]]);
      report.push(name + ": created sheet and headers");
      return;
    }
    const before = sheet.getLastColumn();
    const oldHeaders = before ? sheet.getRange(1, 1, 1, before).getDisplayValues()[0].map(v => String(v || "").trim()) : [];
    const missing = required[name].filter(h => !oldHeaders.includes(h));
    ensureHeaders_(sheet, required[name]);
    report.push(name + (missing.length ? ": appended " + missing.join(", ") : ": headers OK"));
  });
  Logger.log(report.join("\n"));
  return report;
}

function headerMap_(sheet) {
  const lastCol = sheet.getLastColumn();
  if (!lastCol) throw new Error("Sheet " + sheet.getName() + " has no headers.");
  const headers = sheet.getRange(1, 1, 1, lastCol).getDisplayValues()[0].map(v => String(v || "").trim());
  const map = {};
  headers.forEach((h, i) => { if (h) map[h] = i; });
  return { headers: headers, map: map };
}
function rowsAsObjects_(sheet) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];
  const hm = headerMap_(sheet);
  return sheet.getRange(2, 1, lastRow - 1, hm.headers.length).getDisplayValues().map((row, idx) => {
    const obj = { _row: idx + 2, _sheet: sheet.getName() };
    hm.headers.forEach((h, i) => { if (h) obj[h] = row[i]; });
    return obj;
  }).filter(obj => Object.keys(obj).some(k => !k.startsWith("_") && String(obj[k] || "").trim() !== ""));
}
function writeObjectRow_(sheet, object, rowNumber) {
  const hm = headerMap_(sheet);
  const values = hm.headers.map(h => Object.prototype.hasOwnProperty.call(object, h) ? object[h] : "");
  if (rowNumber) sheet.getRange(rowNumber, 1, 1, hm.headers.length).setValues([values]);
  else sheet.getRange(sheet.getLastRow() + 1, 1, 1, hm.headers.length).setValues([values]);
}
function getConfigData() {
  const sheet = getSpreadsheet_().getSheetByName("Config");
  if (!sheet) throw new Error("Config sheet not found.");
  const values = sheet.getDataRange().getDisplayValues();
  if (!values.length) throw new Error("Config sheet is empty.");
  const headers = values[0].map(v => String(v || "").trim());
  const indexes = {};
  headers.forEach((h, i) => indexes[h] = i);
  CONFIG_HEADERS.forEach(h => { if (indexes[h] === undefined) throw new Error("Missing Config column: " + h); });
  const seen = new Set();
  const runners = [];
  values.slice(1).forEach(row => {
    const bib = String(row[indexes["Bib Number"]] || "").trim();
    if (!bib) return;
    if (seen.has(bib)) throw new Error("Duplicate bib number in Config: " + bib);
    seen.add(bib);
    runners.push({ bibNumber: bib, name: String(row[indexes.Name] || "").trim(), category: String(row[indexes.Category] || "").trim(), gender: String(row[indexes.Gender] || "").trim(), distance: detectDistance_(bib) });
  });
  return { success: true, count: runners.length, fetchedAt: new Date().toISOString(), runners: runners };
}
function detectDistance_(bib) {
  const value = String(bib || "").trim();
  if (!/^\d+$/.test(value)) return null;
  if (value.startsWith("16")) return "16KM";
  if (value.startsWith("10")) return "10KM";
  if (value.startsWith("5")) return "5KM";
  if (value.startsWith("3")) return "3KM";
  return null;
}
function uploadRecords(request) {
  if (!Array.isArray(request.records)) throw new Error("records must be an array.");
  if (request.records.length > 200) throw new Error("Upload batches cannot exceed 200 records.");
  const role = String(request.timerRole || "").trim();
  if (!TIMER_SHEETS.includes(role)) throw new Error("Invalid timer role.");
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const sheet = getSpreadsheet_().getSheetByName(role);
    if (!sheet) throw new Error("Missing timer sheet: " + role);
    ensureHeaders_(sheet, TIMER_HEADERS);
    const hm = headerMap_(sheet);
    const existingRows = rowsAsObjects_(sheet);
    const existing = new Map();
    existingRows.forEach(r => { const id = String(r["Record ID"] || "").trim(); if (id) existing.set(id, r); });
    const accepted = [], duplicates = [], conflicts = [], seenRequest = new Set();
    request.records.forEach(record => {
      validateRecord_(record, role);
      const id = String(record.recordId).trim();
      if (seenRequest.has(id)) { duplicates.push(id); return; }
      seenRequest.add(id);
      if (existing.has(id)) {
        const old = existing.get(id);
        const samePayload = String(old["Event ID"] || "").trim() === String(record.eventId || "").trim() && String(old["Bib Number"] || "").trim() === String(record.bibNumber).trim() && String(old.Distance || "").trim() === String(record.distance).trim() && String(old["Finish Timestamp"] || "").trim() === String(record.finishTimestamp).trim();
        if (samePayload) duplicates.push(id); else conflicts.push(id);
        return;
      }
      accepted.push(record);
    });
    if (accepted.length) {
      const uploadedAt = new Date();
      const newRows = accepted.map(record => {
        const o = { "Record ID": String(record.recordId), "Event ID": String(record.eventId || ""), "Timer Role": role, "Bib Number": String(record.bibNumber), "Distance": String(record.distance), "Finish Timestamp": String(record.finishTimestamp), "Elapsed Time": String(record.elapsedTime || ""), "Entry Timestamp": String(record.entryTimestamp || ""), "Record Status": "normal", "Upload Timestamp": uploadedAt, "Device Session ID": String(record.deviceSessionId || ""), "Notes": String(record.notes || "") };
        return hm.headers.map(h => o[h] === undefined ? "" : o[h]);
      });
      sheet.getRange(sheet.getLastRow() + 1, 1, newRows.length, hm.headers.length).setValues(newRows);
    }
    return { success: conflicts.length === 0, accepted: accepted.map(r => r.recordId), duplicates: duplicates, conflicts: conflicts, serverTime: new Date().toISOString() };
  } finally { lock.releaseLock(); }
}
function validateRecord_(record, role) {
  if (!record || typeof record !== "object") throw new Error("Invalid record.");
  if (!record.recordId) throw new Error("Record ID is required.");
  if (record.timerRole && record.timerRole !== role) throw new Error("Record timer role does not match upload.");
  if (!/^\d+$/.test(String(record.bibNumber || ""))) throw new Error("Invalid bib number.");
  const d = detectDistance_(record.bibNumber);
  if (!d) throw new Error("Cannot determine distance for bib " + record.bibNumber);
  if (record.distance !== d) throw new Error("Distance mismatch for bib " + record.bibNumber);
  if (!record.finishTimestamp || isNaN(Date.parse(record.finishTimestamp))) throw new Error("Invalid finish timestamp.");
}
function parseElapsedMs_(value) {
  const s = String(value || "").trim();
  const m = s.match(/^(\d{1,3}):(\d{2}):(\d{2})(?:\.(\d{1,3}))?$/);
  if (!m) return null;
  const frac = Number((m[4] || "0").padEnd(3, "0"));
  return ((Number(m[1]) * 3600 + Number(m[2]) * 60 + Number(m[3])) * 1000) + frac;
}

/** Format elapsed duration as HH:MM:SS.ss (hundredths, not milliseconds). */
function formatElapsedText_(elapsedMs) {
  if (!Number.isFinite(elapsedMs) || elapsedMs < 0) return "";
  const totalCentiseconds = Math.floor(elapsedMs / 10);
  const hours = Math.floor(totalCentiseconds / 360000);
  const minutes = Math.floor((totalCentiseconds % 360000) / 6000);
  const seconds = Math.floor((totalCentiseconds % 6000) / 100);
  const hundredths = totalCentiseconds % 100;
  return [String(hours).padStart(2, "0"), String(minutes).padStart(2, "0"), String(seconds).padStart(2, "0")].join(":") + "." + String(hundredths).padStart(2, "0");
}
function validDefaultCandidate_(record) {
  return Boolean(record && !record.excluded && record.registered &&
    Number.isFinite(record.elapsedMs) && record.elapsedMs >= 0 &&
    record.distance && detectDistance_(record.bibNumber) === record.distance &&
    record.finishTimestamp && !isNaN(Date.parse(record.finishTimestamp)) &&
    record.recordId);
}
function resultKey_(eventId, distance, bib) { return [eventId || "", distance || "", bib || ""].join("|"); }

function getLatestExclusions_() {
  const sheet = ensureSheet_(CORRECTION_SHEET, CORRECTION_HEADERS);
  const logs = rowsAsObjects_(sheet);
  const excluded = new Set();
  logs.forEach(r => {
    const id = String(r["Record ID"] || "").trim();
    if (!id) return;
    if (String(r.Action || "") === "EXCLUDE_RECORD") excluded.add(id);
    if (String(r.Action || "") === "RESTORE_RECORD") excluded.delete(id);
  });
  return excluded;
}
function getLatestBibCorrections_() {
  const sheet = ensureSheet_(CORRECTION_SHEET, CORRECTION_HEADERS);
  const logs = rowsAsObjects_(sheet);
  const corrections = new Map();
  logs.forEach(r => {
    if (String(r.Action || "") !== "CORRECT_BIB") return;
    const id = String(r["Record ID"] || "").trim();
    const newBib = String(r["New Value"] || "").trim();
    if (id && newBib) corrections.set(id, newBib);
  });
  return corrections;
}

function getReviewData(eventIdFilter) {
  const ss = getSpreadsheet_();
  const config = getConfigData();
  const runnerMap = new Map(config.runners.map(r => [String(r.bibNumber), r]));
  const exclusions = getLatestExclusions_();
  const corrections = getLatestBibCorrections_();
  const all = [];
  TIMER_SHEETS.forEach(role => {
    const sheet = ss.getSheetByName(role);
    if (!sheet) throw new Error("Missing timer sheet: " + role);
    ensureHeaders_(sheet, TIMER_HEADERS);
    rowsAsObjects_(sheet).forEach(row => {
      const eventId = String(row["Event ID"] || "").trim();
      if (eventIdFilter && eventId !== eventIdFilter) return;
      const recordId = String(row["Record ID"] || "").trim();
      const rawBib = String(row["Bib Number"] || "").trim();
      const bib = corrections.get(recordId) || rawBib;
      const distance = String(row.Distance || detectDistance_(rawBib) || "").trim();
      const runner = runnerMap.get(bib) || null;
      const rawElapsed = String(row["Elapsed Time"] || "").trim();
      const elapsedMs = parseElapsedMs_(rawElapsed);
      const elapsed = elapsedMs === null ? rawElapsed : formatElapsedText_(elapsedMs);
      all.push({ recordId: recordId, eventId: eventId, timerRole: String(row["Timer Role"] || role), rawBibNumber: rawBib, bibNumber: bib, bibCorrected: bib !== rawBib, distance: distance, finishTimestamp: String(row["Finish Timestamp"] || ""), elapsedTime: elapsed, elapsedMs: elapsedMs, entryTimestamp: String(row["Entry Timestamp"] || ""), recordStatus: String(row["Record Status"] || "normal"), notes: String(row.Notes || ""), runnerName: runner ? runner.name : "", category: runner ? runner.category : "", gender: runner ? runner.gender : "", registered: Boolean(runner), excluded: exclusions.has(recordId), sheetName: role });
    });
  });

  const groups = new Map();
  all.forEach(r => {
    const key = resultKey_(r.eventId, r.distance, r.bibNumber);
    if (!groups.has(key)) groups.set(key, []);
    groups.get(key).push(r);
  });
  all.forEach(r => {
    const issues = [];
    if (!r.registered) issues.push("INVALID_BIB");
    if (!r.bibNumber || !/^\d+$/.test(r.bibNumber)) issues.push("INVALID_BIB_FORMAT");
    if (!r.distance || detectDistance_(r.bibNumber) !== r.distance) issues.push("DISTANCE_MISMATCH");
    if (!r.finishTimestamp || isNaN(Date.parse(r.finishTimestamp))) issues.push("INVALID_FINISH_TIMESTAMP");
    if (r.elapsedMs === null || r.elapsedMs < 0) issues.push("INVALID_OR_MISSING_ELAPSED_TIME");
    const group = (groups.get(resultKey_(r.eventId, r.distance, r.bibNumber)) || []).filter(x => !x.excluded);
    if (!r.excluded && group.filter(x => x.timerRole === r.timerRole).length > 1) issues.push("DUPLICATE_CANDIDATE");
    if (!r.excluded && !group.some(x => x.timerRole === "Timer1") && group.some(x => x.timerRole !== "Timer1")) issues.push("MISSING_PRIMARY");
    if (r.excluded) issues.push("EXCLUDED");
    r.issues = [...new Set(issues)];
    r.status = r.excluded ? "EXCLUDED" : (r.issues.length ? r.issues.join(", ") : "VALID");
  });

  syncTimer1Defaults_(all);
  const consolidatedSheet = ensureSheet_(CONSOLIDATED_SHEET, CONSOLIDATED_HEADERS);
  const consolidated = rowsAsObjects_(consolidatedSheet)
    .filter(r => !eventIdFilter || String(r["Event ID"] || "") === eventIdFilter)
    .map(r => {
      const rawElapsed = String(r["Elapsed Time"] || "").trim();
      const elapsedMs = parseElapsedMs_(rawElapsed);
      return { resultKey: r["Result Key"], eventId: r["Event ID"], distance: r.Distance, bibNumber: r["Bib Number"], name: r.Name, category: r.Category, gender: r.Gender, selectedRecordId: r["Selected Record ID"], selectedTimer: r["Selected Timer"], finishTimestamp: r["Finish Timestamp"], elapsedTime: elapsedMs === null ? rawElapsed : formatElapsedText_(elapsedMs), selectionStatus: r["Selection Status"], notes: r.Notes, updatedAt: r["Updated At"], updatedBy: r["Updated By"] };
    });
  const distances = [...new Set(all.map(r => r.distance).filter(Boolean))].sort();
  const eventIds = [...new Set(all.map(r => r.eventId).filter(Boolean))].sort();
  return { success: true, fetchedAt: new Date().toISOString(), count: all.length, distances: distances, eventIds: eventIds, records: all, consolidated: consolidated, categories: [...new Set(config.runners.map(r => r.category).filter(Boolean))].sort(), genders: [...new Set(config.runners.map(r => r.gender).filter(Boolean))].sort() };
}

/**
 * Default selection rules on refresh:
 * - If exactly one active Timer1 observation exists and is valid, select it provisionally.
 * - If Timer1 has multiple observations for the bib, leave the selection unresolved for manual review.
 * - If there is no active Timer1 observation, select the fastest valid backup observation as a suggestion,
 *   but mark it NEEDS REVIEW (never FINAL).
 * - Never overwrite a FINAL row or a human-confirmed selection.
 */
function syncTimer1Defaults_(records) {
  const sheet = ensureSheet_(CONSOLIDATED_SHEET, CONSOLIDATED_HEADERS);
  const lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    const rows = rowsAsObjects_(sheet);
    const byKey = new Map();
    rows.forEach(row => {
      const key = String(row["Result Key"] || "");
      if (key && !byKey.has(key)) byKey.set(key, row);
    });

    const groups = new Map();
    records.forEach(record => {
      const key = resultKey_(record.eventId, record.distance, record.bibNumber);
      if (!groups.has(key)) groups.set(key, []);
      groups.get(key).push(record);
    });

    const isAutoSource = value => /^(AUTO:|Timer1 default$|Fastest backup default$)/i.test(String(value || "").trim());

    groups.forEach((group, key) => {
      const existing = byKey.get(key) || null;
      const existingStatus = String(existing && existing["Selection Status"] || "").trim().toUpperCase();
      const selectedId = String(existing && existing["Selected Record ID"] || "").trim();
      const updatedBy = String(existing && existing["Updated By"] || "").trim();

      // Never rewrite finalized results. Preserve explicit group exclusions until the director restores/selects.
      if (existingStatus === "FINAL") return;
      if (existingStatus === "EXCLUDED" && !selectedId) return;

      // Do not overwrite a human-confirmed result. Automatically generated suggestions can be recomputed.
      if (selectedId && !isAutoSource(updatedBy)) return;

      const active = group.filter(record => !record.excluded);
      if (!active.length) return;

      const timer1Observations = active.filter(record => record.timerRole === "Timer1");
      let candidate = null;
      let selectionStatus = "NEEDS REVIEW";
      let updatedByValue = "AUTO: REVIEW REQUIRED";
      let notes = "Review required before finalization.";

      if (timer1Observations.length === 1) {
        const timer1Candidate = timer1Observations[0];
        if (validDefaultCandidate_(timer1Candidate)) {
          candidate = timer1Candidate;
          selectionStatus = "PROVISIONAL";
          updatedByValue = "AUTO: TIMER1 DEFAULT";
          notes = "Timer1 default; provisional pending review/finalization.";
        } else {
          notes = "Timer1 observation is invalid or incomplete; review required.";
        }
      } else if (timer1Observations.length > 1) {
        notes = "Multiple Timer1 observations for this bib; choose the correct observation manually.";
      } else {
        // Timer1 has no active observation: choose the fastest valid backup only as a reviewable suggestion.
        const backups = active.filter(record => record.timerRole !== "Timer1" && validDefaultCandidate_(record));
        backups.sort((a, b) => {
          if (a.elapsedMs !== b.elapsedMs) return a.elapsedMs - b.elapsedMs;
          const aFinish = Date.parse(a.finishTimestamp) || 0;
          const bFinish = Date.parse(b.finishTimestamp) || 0;
          if (aFinish !== bFinish) return aFinish - bFinish;
          return String(a.timerRole).localeCompare(String(b.timerRole));
        });
        if (backups.length) {
          candidate = backups[0];
          selectionStatus = "NEEDS REVIEW";
          updatedByValue = "AUTO: FASTEST BACKUP";
          notes = "Fastest valid backup suggested because Timer1 has no active observation. Review before finalization.";
        } else {
          notes = "No valid observation available for automatic selection.";
        }
      }

      // A previously auto-selected result must be cleared if Timer1 duplicates make the choice ambiguous.
      const dataRecord = candidate || group.find(record => !record.excluded) || group[0];
      const object = {
        "Result Key": key,
        "Event ID": dataRecord ? dataRecord.eventId : "",
        "Distance": dataRecord ? dataRecord.distance : "",
        "Bib Number": dataRecord ? dataRecord.bibNumber : "",
        "Name": candidate ? candidate.runnerName : (dataRecord ? dataRecord.runnerName : ""),
        "Category": candidate ? candidate.category : (dataRecord ? dataRecord.category : ""),
        "Gender": candidate ? candidate.gender : (dataRecord ? dataRecord.gender : ""),
        "Selected Record ID": candidate ? candidate.recordId : "",
        "Selected Timer": candidate ? candidate.timerRole : "",
        "Finish Timestamp": candidate ? candidate.finishTimestamp : "",
        "Elapsed Time": candidate ? formatElapsedText_(candidate.elapsedMs) : "",
        "Selection Status": selectionStatus,
        "Notes": notes,
        "Updated At": new Date(),
        "Updated By": updatedByValue
      };

      if (existing) {
        writeObjectRow_(sheet, object, existing._row);
        byKey.set(key, { ...object, _row: existing._row });
      } else {
        writeObjectRow_(sheet, object);
        byKey.set(key, { ...object, _row: sheet.getLastRow() });
      }
    });
  } finally {
    lock.releaseLock();
  }
}

function appendCorrectionLog_(action, record, oldValue, newValue, reason, actor) {
  const sheet = ensureSheet_(CORRECTION_SHEET, CORRECTION_HEADERS);
  const logId = Utilities.getUuid();
  const timestamp = new Date();
  const key = resultKey_(record.eventId || "", record.distance || "", record.bibNumber || "");
  const values = {
    "Log ID": logId, "Correction ID": logId, "Action": action, "Event ID": record.eventId || "",
    "Distance": record.distance || "", "Bib Number": record.bibNumber || "", "Record ID": record.recordId || "",
    "Timer Role": record.timerRole || "", "Previous Value": oldValue == null ? "" : String(oldValue),
    "Original Value": oldValue == null ? "" : String(oldValue), "New Value": newValue == null ? "" : String(newValue),
    "Reason": reason || "", "Actor": actor || "Timer1", "Timestamp": timestamp, "Correction Timestamp": timestamp,
    "Consolidated ID": key, "Field Changed": action, "Device Session ID": record.deviceSessionId || ""
  };
  writeObjectRow_(sheet, values);
}

function selectResults_(request) {
  const ids = [...new Set((request.recordIds || []).map(String).filter(Boolean))];
  if (!ids.length) throw new Error("Select at least one result.");
  if (ids.length > 200) throw new Error("Select up to 200 results at a time.");
  const data = getReviewData(String(request.eventId || ""));
  const records = ids.map(id => data.records.find(r => r.recordId === id)).filter(Boolean);
  if (records.length !== ids.length) throw new Error("One or more records could not be found. Refresh and try again.");
  records.forEach(record => {
    if (record.excluded) throw new Error("An excluded record cannot be selected. Restore it first.");
    if (record.elapsedMs === null || record.elapsedMs < 0 || record.issues.includes("INVALID_BIB") || record.issues.includes("DISTANCE_MISMATCH") || record.issues.includes("INVALID_FINISH_TIMESTAMP")) {
      throw new Error("Correct the invalid bib, distance, or time before selecting this result.");
    }
  });
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    const sheet = ensureSheet_(CONSOLIDATED_SHEET, CONSOLIDATED_HEADERS);
    const rows = rowsAsObjects_(sheet);
    records.forEach(record => {
      const key = resultKey_(record.eventId, record.distance, record.bibNumber);
      const existing = rows.find(r => String(r["Result Key"] || "") === key);
      const isBackupSuggestion = request.applyDefaults === true && record.timerRole !== "Timer1";
      const obj = { "Result Key": key, "Event ID": record.eventId, "Distance": record.distance, "Bib Number": record.bibNumber, "Name": record.runnerName, "Category": record.category, "Gender": record.gender, "Selected Record ID": record.recordId, "Selected Timer": record.timerRole, "Finish Timestamp": record.finishTimestamp, "Elapsed Time": formatElapsedText_(record.elapsedMs), "Selection Status": isBackupSuggestion ? "NEEDS REVIEW" : "PROVISIONAL", "Notes": isBackupSuggestion ? "Fastest backup suggestion; review required before finalization." : "", "Updated At": new Date(), "Updated By": isBackupSuggestion ? "AUTO: FASTEST BACKUP" : "Timer1" };
      if (existing) writeObjectRow_(sheet, obj, existing._row); else writeObjectRow_(sheet, obj);
      appendCorrectionLog_("SELECT_RESULT", record, existing ? existing["Selected Record ID"] : "", record.recordId, "", "Timer1");
    });
    return { success: true, message: "Selected " + records.length + " result(s).", count: records.length };
  } finally { lock.releaseLock(); }
}

function excludeRecords_(request) {
  const ids = [...new Set((request.recordIds || []).map(String).filter(Boolean))];
  if (!ids.length) throw new Error("Select at least one observation to exclude.");
  if (ids.length > 500) throw new Error("Exclude up to 500 observations at a time.");
  const data = getReviewData(String(request.eventId || ""));
  const records = ids.map(id => data.records.find(r => r.recordId === id)).filter(Boolean);
  const recordMap = new Map(records.map(r => [r.recordId, r]));
  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    records.forEach(record => { if (!record.excluded) appendCorrectionLog_("EXCLUDE_RECORD", record, "included", "excluded", "", "Timer1"); });
    const sheet = ensureSheet_(CONSOLIDATED_SHEET, CONSOLIDATED_HEADERS);
    const rows = rowsAsObjects_(sheet);
    let changed = 0;
    rows.forEach(row => {
      const selectedId = String(row["Selected Record ID"] || "");
      if (selectedId && recordMap.has(selectedId)) {
        const obj = { "Result Key": row["Result Key"], "Event ID": row["Event ID"], "Distance": row.Distance, "Bib Number": row["Bib Number"], "Name": row.Name, "Category": row.Category, "Gender": row.Gender, "Selected Record ID": "", "Selected Timer": "", "Finish Timestamp": "", "Elapsed Time": "", "Selection Status": "EXCLUDED", "Notes": "", "Updated At": new Date(), "Updated By": "Timer1" };
        writeObjectRow_(sheet, obj, row._row);
        changed++;
      }
    });
    return { success: true, message: "Excluded " + records.filter(r => !r.excluded).length + " observation(s). " + changed + " consolidated selection(s) cleared.", count: records.filter(r => !r.excluded).length };
  } finally { lock.releaseLock(); }
}

function restoreRecords_(request) {
  const ids = [...new Set((request.recordIds || []).map(String).filter(Boolean))];
  if (!ids.length) throw new Error("Select at least one excluded observation to restore.");
  const data = getReviewData(String(request.eventId || ""));
  const records = ids.map(id => data.records.find(r => r.recordId === id)).filter(Boolean);
  const restored = records.filter(record => record.excluded);
  restored.forEach(record => appendCorrectionLog_("RESTORE_RECORD", record, "excluded", "included", "", "Timer1"));

  // Release the prior group-level EXCLUDED marker; the next refresh calculates a valid default again.
  if (restored.length) {
    const sheet = ensureSheet_(CONSOLIDATED_SHEET, CONSOLIDATED_HEADERS);
    const restoredKeys = new Set(restored.map(record => resultKey_(record.eventId, record.distance, record.bibNumber)));
    rowsAsObjects_(sheet).forEach(row => {
      const key = String(row["Result Key"] || "");
      if (restoredKeys.has(key) && String(row["Selection Status"] || "").toUpperCase() === "EXCLUDED") {
        const obj = { "Result Key": key, "Event ID": row["Event ID"], "Distance": row.Distance, "Bib Number": row["Bib Number"], "Name": row.Name, "Category": row.Category, "Gender": row.Gender, "Selected Record ID": "", "Selected Timer": "", "Finish Timestamp": "", "Elapsed Time": "", "Selection Status": "NEEDS REVIEW", "Notes": "Observation restored; default selection recalculated.", "Updated At": new Date(), "Updated By": "AUTO: REVIEW REQUIRED" };
        writeObjectRow_(sheet, obj, row._row);
      }
    });
  }
  return { success: true, message: "Restored " + restored.length + " observation(s) to review.", count: restored.length };
}

function correctBib_(request) {
  const ids = [...new Set((request.recordIds || [request.recordId]).map(String).filter(Boolean))];
  const newBib = String(request.newBibNumber || "").trim();
  if (!ids.length || !/^\d+$/.test(newBib)) throw new Error("Enter a valid numeric registered bib.");
  const data = getReviewData(String(request.eventId || ""));
  const targets = ids.map(id => data.records.find(r => r.recordId === id));
  if (targets.some(r => !r)) throw new Error("One or more observations were not found. Refresh and try again.");
  const runner = getConfigData().runners.find(r => String(r.bibNumber) === newBib);
  if (!runner) throw new Error("Bib " + newBib + " is not in Config.");
  targets.forEach(r => {
    if (detectDistance_(newBib) !== r.distance) throw new Error("Bib " + newBib + " belongs to " + detectDistance_(newBib) + ", not " + r.distance + ".");
  });
  const sheet = ensureSheet_(CONSOLIDATED_SHEET, CONSOLIDATED_HEADERS);
  const consolidated = rowsAsObjects_(sheet);
  const selectedTarget = consolidated.find(row => ids.includes(String(row["Selected Record ID"] || "")));
  if (selectedTarget) {
    const destinationKey = resultKey_(selectedTarget["Event ID"], selectedTarget.Distance, newBib);
    const collision = consolidated.find(row => String(row["Result Key"] || "") === destinationKey && row._row !== selectedTarget._row && String(row["Selected Record ID"] || "").trim());
    if (collision) throw new Error("This bib already has a selected consolidated result. Resolve that result first.");
  }
  targets.forEach(record => appendCorrectionLog_("CORRECT_BIB", record, record.bibNumber, newBib, "", "Timer1"));
  // If a corrected observation was already selected, move/update only that consolidated row.
  if (selectedTarget) {
    const chosenRecord = targets.find(r => r.recordId === String(selectedTarget["Selected Record ID"] || ""));
    const correctedRow = { "Result Key": resultKey_(chosenRecord.eventId, chosenRecord.distance, newBib), "Event ID": chosenRecord.eventId, "Distance": chosenRecord.distance, "Bib Number": newBib, "Name": runner.name, "Category": runner.category, "Gender": runner.gender, "Selected Record ID": chosenRecord.recordId, "Selected Timer": chosenRecord.timerRole, "Finish Timestamp": chosenRecord.finishTimestamp, "Elapsed Time": chosenRecord.elapsedTime, "Selection Status": "PROVISIONAL", "Notes": "", "Updated At": new Date(), "Updated By": "Timer1" };
    writeObjectRow_(sheet, correctedRow, selectedTarget._row);
  }
  return { success: true, message: "Corrected " + targets.length + " observation(s) to bib " + newBib + ".", count: targets.length, bibNumber: newBib };
}

function finalizeResults_(request) {
  const eventId = String(request.eventId || "").trim();
  if (!eventId) throw new Error("Select one event before finalizing.");

  // Refresh to calculate defaults for records that arrived since the previous review.
  getReviewData(eventId);
  const sheet = ensureSheet_(CONSOLIDATED_SHEET, CONSOLIDATED_HEADERS);
  const hm = headerMap_(sheet);
  const rows = rowsAsObjects_(sheet).filter(r => String(r["Event ID"] || "") === eventId);
  if (!rows.length) throw new Error("No consolidated results exist for this event.");

  const unresolved = rows.filter(row => {
    const status = String(row["Selection Status"] || "").trim().toUpperCase();
    const selectedId = String(row["Selected Record ID"] || "").trim();
    return status === "NEEDS REVIEW" || (!selectedId && status !== "EXCLUDED" && status !== "FINAL");
  });
  if (unresolved.length) {
    const bibs = unresolved.map(row => String(row["Bib Number"] || "?")).slice(0, 12);
    throw new Error("Review or exclude these bib results before finalizing: " + bibs.join(", ") + (unresolved.length > 12 ? " and " + (unresolved.length - 12) + " more" : ""));
  }

  const selectedRows = rows.filter(row => String(row["Selected Record ID"] || "").trim());
  if (!selectedRows.length) throw new Error("No selected consolidated results exist for this event.");

  const lock = LockService.getScriptLock(); lock.waitLock(20000);
  try {
    selectedRows.forEach(row => {
      const statusCol = hm.map["Selection Status"] + 1;
      const notesCol = hm.map.Notes + 1;
      const updatedCol = hm.map["Updated At"] + 1;
      const actorCol = hm.map["Updated By"] + 1;
      sheet.getRange(row._row, statusCol).setValue("FINAL");
      sheet.getRange(row._row, notesCol).setValue("");
      sheet.getRange(row._row, updatedCol).setValue(new Date());
      sheet.getRange(row._row, actorCol).setValue("Timer1");
      appendCorrectionLog_("FINALIZE_EVENT", { eventId: eventId, distance: row.Distance, bibNumber: row["Bib Number"], recordId: row["Selected Record ID"], timerRole: row["Selected Timer"] }, row["Selection Status"], "FINAL", "", "Timer1");
    });
    return { success: true, message: "Finalized " + selectedRows.length + " consolidated result(s) for " + eventId + ".", count: selectedRows.length };
  } finally { lock.releaseLock(); }
}

function createId() {
    if (crypto.randomUUID) {
        return crypto.randomUUID();
    }

    return "v5000-" + Date.now() + "-" +
        Math.random().toString(36).slice(2);
}

function nowIso() {
    return new Date().toISOString();
}

function normalizeBib(value) {
    return String(value ?? "").trim();
}

function formatClock(iso) {
    if (!iso) return "--:--:--";

    return new Intl.DateTimeFormat("en-GB", {
        timeZone: "Asia/Manila",
        hour: "2-digit",
        minute: "2-digit",
        second: "2-digit",
        hour12: false
    }).format(new Date(iso));
}

function formatElapsed(ms) {
    if (!Number.isFinite(ms) || ms < 0) return "Pending";

    const centiseconds = Math.floor(ms / 10);
    const hours = Math.floor(centiseconds / 360000);
    const minutes = Math.floor((centiseconds % 360000) / 6000);
    const seconds = Math.floor((centiseconds % 6000) / 100);
    const fraction = centiseconds % 100;

    return [
        String(hours).padStart(2, "0"),
        String(minutes).padStart(2, "0"),
        String(seconds).padStart(2, "0")
    ].join(":") + "." + String(fraction).padStart(2, "0");
}

/*
 * Longer prefixes must be checked first.
 * 16001 must be 16KM, not 3KM.
 */
function detectDistance(bib) {
    const value = normalizeBib(bib);

    if (!/^\d+$/.test(value)) return null;

    if (value.startsWith("16")) return "16KM";
    if (value.startsWith("10")) return "10KM";
    if (value.startsWith("5")) return "5KM";
    if (value.startsWith("3")) return "3KM";

    return null;
}

function openDatabase() {
    return new Promise((resolve, reject) => {
        const request = indexedDB.open(DB_NAME, DB_VERSION);

        request.onupgradeneeded = event => {
            const database = event.target.result;

            if (!database.objectStoreNames.contains(STORE_ENTRIES)) {
                const store = database.createObjectStore(STORE_ENTRIES, {
                    keyPath: "recordId"
                });

                store.createIndex("eventId", "eventId");
                store.createIndex("timerRole", "timerRole");
                store.createIndex("distance", "distance");
            }

            if (!database.objectStoreNames.contains(STORE_META)) {
                database.createObjectStore(STORE_META, {
                    keyPath: "key"
                });
            }

            if (!database.objectStoreNames.contains(STORE_CONFIG)) {
                database.createObjectStore(STORE_CONFIG, {
                    keyPath: "key"
                });
            }
        };

        request.onsuccess = () => {
            db = request.result;
            resolve(db);
        };

        request.onerror = () => reject(request.error);
    });
}

function getFromStore(storeName, key) {
    return new Promise((resolve, reject) => {
        const transaction = db.transaction(storeName, "readonly");
        const request = transaction.objectStore(storeName).get(key);

        request.onsuccess = () => {
            resolve(request.result ? request.result.value : undefined);
        };

        request.onerror = () => reject(request.error);
    });
}

function getAllFromStore(storeName) {
    return new Promise((resolve, reject) => {
        const transaction = db.transaction(storeName, "readonly");
        const request = transaction.objectStore(storeName).getAll();

        request.onsuccess = () => resolve(request.result || []);
        request.onerror = () => reject(request.error);
    });
}

function putInStore(storeName, key, value) {
    return new Promise((resolve, reject) => {
        const transaction = db.transaction(storeName, "readwrite");
        const store = transaction.objectStore(storeName);

        const request = storeName === STORE_ENTRIES
            ? store.put(value)
            : store.put({ key, value });

        transaction.oncomplete = () => resolve(request.result);
        transaction.onerror = () => reject(transaction.error);
        transaction.onabort = () => reject(transaction.error);
    });
}

function addEntryToStore(entry) {
    return new Promise((resolve, reject) => {
        const transaction = db.transaction(STORE_ENTRIES, "readwrite");
        const request = transaction.objectStore(STORE_ENTRIES).add(entry);

        transaction.oncomplete = () => resolve();
        transaction.onerror = () => reject(transaction.error);
        transaction.onabort = () => reject(transaction.error);
    });
}

async function persistState() {
    const savedState = {
        eventName: state.eventName,
        eventId: state.eventId,
        timerRole: state.timerRole,
        deviceLabel: state.deviceLabel,
        selectedDistance: state.selectedDistance,
        starts: state.starts,
        startHistory: state.startHistory || {},
        screen: state.screen,
        deviceSessionId: state.deviceSessionId
    };

    await putInStore(STORE_META, "appState", savedState);
}

async function restoreState() {
    const savedState = await getFromStore(STORE_META, "appState");

    if (savedState) {
        state = { ...state, ...savedState };
    }

    if (!state.startHistory || typeof state.startHistory !== "object") state.startHistory = {};

    if (!state.deviceSessionId) {
        state.deviceSessionId = createId();
    }

    state.entries = await getAllFromStore(STORE_ENTRIES);
    state.entries.forEach(entry => { if (entry.reviewRequired === undefined) entry.reviewRequired = false; });

    const cachedConfig = await getFromStore(
        STORE_CONFIG,
        CONFIG_CACHE_KEY
    );

    if (cachedConfig && Array.isArray(cachedConfig.runners)) {
        state.runners = cachedConfig.runners;
        state.configFetchedAt = cachedConfig.fetchedAt || null;
    }

    if (!["setup", "timing", "results"].includes(state.screen)) {
        state.screen = "setup";
    }
}

async function refreshConfig() {
    if (configRefreshing || !navigator.onLine) return;

    configRefreshing = true;
    $("configStatus").textContent = "Refreshing…";

    try {
        const response = await fetch(API_URL + "?action=config", {
            method: "GET",
            cache: "no-store",
            redirect: "follow"
        });

        if (!response.ok) {
            throw new Error("Config request failed: HTTP " + response.status);
        }

        const data = await response.json();

        if (!data.success || !Array.isArray(data.runners)) {
            throw new Error(data.error || "Invalid Config response.");
        }

        const seen = new Set();

        for (const runner of data.runners) {
            const bib = normalizeBib(runner.bibNumber);

            if (!bib || seen.has(bib)) {
                throw new Error("Config contains a blank or duplicate bib: " + bib);
            }

            seen.add(bib);

            // Recalculate locally instead of blindly trusting the server value.
            runner.bibNumber = bib;
            runner.distance = detectDistance(bib);
        }

        await putInStore(STORE_CONFIG, CONFIG_CACHE_KEY, {
            runners: data.runners,
            fetchedAt: data.fetchedAt || nowIso()
        });

        state.runners = data.runners;
        state.configFetchedAt = data.fetchedAt || nowIso();

        $("configStatus").textContent =
            state.runners.length + " runners";

        setConnectionStatus("CONNECTED");
        renderBibLookup();

        setSyncMessage(
            "Config synchronized: " + state.runners.length +
            " runners. Finish records are still saved locally before upload."
        );

    } catch (error) {
        console.error("Config refresh failed:", error);

        $("configStatus").textContent = state.runners.length
            ? state.runners.length + " cached"
            : "Unavailable";

        setConnectionStatus(navigator.onLine ? "SYNC ISSUE" : "OFFLINE");

        setSyncMessage(
            state.runners.length
                ? "Using cached Config. New records can still be saved locally."
                : "Config is not available yet. Records can be saved locally, but registration cannot be verified."
        );

    } finally {
        configRefreshing = false;
    }
}

function parseManilaLocalDateTime(value) {
    // datetime-local values are interpreted in the device's local timezone.
    // Race devices should be configured to Asia/Manila.
    const parsed = new Date(value);
    if (!value || Number.isNaN(parsed.getTime())) return null;
    return parsed.toISOString();
}

async function editStartTime(distance) {
    const current = state.starts[distance];
    if (!current) return;

    const currentLocal = (() => {
        const date = new Date(current);
        const pad = n => String(n).padStart(2, "0");
        return date.getFullYear() + "-" + pad(date.getMonth() + 1) + "-" +
            pad(date.getDate()) + "T" + pad(date.getHours()) + ":" +
            pad(date.getMinutes()) + ":" + pad(date.getSeconds());
    })();

    const entered = prompt(
        "Edit " + distance + " start time. Enter date/time as YYYY-MM-DDTHH:mm:ss.\n" +
        "Use the timing device's local time (set to Philippines time).\nCurrent: " + currentLocal,
        currentLocal
    );
    if (entered === null) return;

    const correctedIso = parseManilaLocalDateTime(entered.trim());
    if (!correctedIso) {
        alert("Invalid date/time. Use YYYY-MM-DDTHH:mm:ss.");
        return;
    }

    const affected = getEntriesForDistance(distance);
    const invalid = affected.filter(entry =>
        new Date(entry.finishTimestamp).getTime() < new Date(correctedIso).getTime()
    );
    if (invalid.length) {
        const proceed = confirm(
            "Warning: " + invalid.length + " finish record(s) would be earlier than this start time. " +
            "They will be marked for review and show Pending elapsed time. Continue?"
        );
        if (!proceed) return;
    }

    const confirmEdit = confirm(
        "Change the " + distance + " start time to " + entered.trim() + "?\n" +
        affected.length + " local finish record(s) will be recalculated. Original finish timestamps will not change."
    );
    if (!confirmEdit) return;

    if (!state.startHistory[distance]) state.startHistory[distance] = [];
    state.startHistory[distance].push({
        action: "EDIT",
        previousStart: current,
        newStart: correctedIso,
        changedAt: nowIso(),
        recordsAffected: affected.length
    });

    state.starts[distance] = correctedIso;

    for (const entry of affected) {
        const elapsed = new Date(entry.finishTimestamp).getTime() - new Date(correctedIso).getTime();
        entry.startTimestamp = correctedIso;
        entry.elapsedMs = elapsed >= 0 ? elapsed : null;
        entry.elapsedTime = elapsed >= 0 ? formatElapsed(elapsed) : "";
        entry.reviewRequired = true;
        entry.notes = [entry.notes, "Start time edited; review required."].filter(Boolean).join(" ");
        await saveUpdatedEntry(entry);
    }

    await persistState();
    renderStarts();
    renderRecentEntries();
    updateCounters();
    if (state.screen === "results") renderResults();
    showMessage(distance + " start time updated. Affected results are marked for review.", "success");
}

async function undoStart(distance) {
    const current = state.starts[distance];
    if (!current) return;

    const affected = getEntriesForDistance(distance);
    const warning = affected.length
        ? "\n\n" + affected.length + " finish record(s) exist for this distance. Their finish timestamps will be kept, but elapsed time will become Pending until a start time is set again."
        : "";

    if (!confirm("Undo the " + distance + " start time? This only affects " + distance + "." + warning)) {
        return;
    }

    if (!state.startHistory[distance]) state.startHistory[distance] = [];
    state.startHistory[distance].push({
        action: "UNDO",
        previousStart: current,
        newStart: null,
        changedAt: nowIso(),
        recordsAffected: affected.length
    });

    delete state.starts[distance];

    for (const entry of affected) {
        entry.startTimestamp = null;
        entry.elapsedMs = null;
        entry.elapsedTime = "";
        entry.reviewRequired = true;
        entry.notes = [entry.notes, "Start time undone; elapsed time pending; review required."].filter(Boolean).join(" ");
        await saveUpdatedEntry(entry);
    }

    await persistState();
    renderStarts();
    renderRecentEntries();
    updateCounters();
    showMessage(distance + " start undone. Finish records were preserved.", "success");
}

async function startDistance(distance) {

        showMessage(distance + " start saved at " + formatClock(timestamp), "success");
    } catch (error) {

        showMessage("Could not save start time.", "error");
    }
}

async function submitBib() {
    if (submitting) return;

    const bib = normalizeBib($("bibInput").value);

    if (!bib) {
        showMessage("Enter a bib number first.", "error");
        return;
    }

    const distance = detectDistance(bib);

    if (!distance) {
        showMessage(
            "Cannot determine distance from this bib. Record not saved.",
            "error"
        );
        return;
    }

    const runner = findRunner(bib);
    const finishTimestamp = nowIso();
    const startTimestamp = state.starts[distance] || null;

    const elapsedMs = startTimestamp
        ? new Date(finishTimestamp).getTime() -
        new Date(startTimestamp).getTime()
        : null;

    if (elapsedMs !== null && elapsedMs < 0) {
        showMessage(
            "Finish time is earlier than the saved start time. Check device clock.",
            "error"
        );
        return;
    }

    const entry = {
        recordId: createId(),
        eventId: state.eventId,
        eventName: state.eventName,
        timerRole: state.timerRole,
        deviceSessionId: state.deviceSessionId,
        deviceLabel: state.deviceLabel,

        bibNumber: bib,
        distance,

        runnerName: runner ? runner.name : "",
        category: runner ? runner.category : "",
        gender: runner ? runner.gender : "",
        registrationStatus: runner ? "registered" : "unregistered-review",

        finishTimestamp,
        startTimestamp,
        elapsedMs,
        elapsedTime: elapsedMs === null ? "" : formatElapsed(elapsedMs),

        entryTimestamp: finishTimestamp,
        recordStatus: "normal",
        reviewRequired: false,
        syncStatus: "pending",
        notes: runner ? "" : "Bib not found in cached Config; review required."
    };

    submitting = true;

    const enterButton = document.querySelector('[data-key="enter"]');
    enterButton.disabled = true;

    try {
        // Local write completes before we report success.
        await addEntryToStore(entry);

        state.entries.push(entry);

        $("bibInput").value = "";
        renderBibLookup();
        renderRecentEntries();
        updateCounters();

        showMessage(
            "SAVED LOCALLY · Bib " + bib + " · " + distance +
            (entry.elapsedTime ? " · " + entry.elapsedTime : " · Start not recorded"),
            "success"
        );

        await persistState();

    } catch (error) {
        console.error(error);
        showMessage(
            "SAVE FAILED. Do not assume this record was stored. Try again.",
            "error"
        );

    } finally {
        submitting = false;
        enterButton.disabled = false;
        focusBibInput();
    }
}

/*
 * This sends text/plain to avoid a JSON content-type preflight.
 * The Apps Script deployment must be accessible to the browser.
 */
async function postToAppsScript(payload) {
    const response = await fetch(API_URL, {
        method: "POST",
        redirect: "follow",
        headers: {
            "Content-Type": "text/plain;charset=utf-8"
        },
        body: JSON.stringify(payload)
    });

    if (!response.ok) {
        throw new Error("Upload HTTP error: " + response.status);
    }

    const result = await response.json();

    if (!result || typeof result.success !== "boolean") {
        throw new Error("Unexpected upload response.");
    }

    return result;
}

async function saveUpdatedEntry(entry) {
    await putInStore(STORE_ENTRIES, entry.recordId, entry);

    const index = state.entries.findIndex(
        item => item.recordId === entry.recordId
    );

    if (index >= 0) {
        state.entries[index] = entry;
    }
}

async function submitPendingRecords() {
    if (submitting) return;
    const pending = state.entries.filter(entry => entry.syncStatus !== "synced");
    if (!pending.length) { alert("There are no pending records to upload."); return; }
    if (!navigator.onLine) { alert("You are offline. Records remain saved locally."); return; }

    submitting = true;
    $("submitButton").disabled = true;
    setConnectionStatus("UPLOADING");
    setSyncMessage("Uploading pending records by their saved timer role. Keep the app open.");

    let confirmedCount = 0;
    const failures = [];
    try {
        const groups = new Map();
        for (const entry of pending) {
            const role = String(entry.timerRole || "").trim();
            if (!/^Timer[1-5]$/.test(role)) {
                failures.push("Bib " + entry.bibNumber + ": invalid saved timer role.");
                continue;
            }
            if (!groups.has(role)) groups.set(role, []);
            groups.get(role).push(entry);
        }

        for (const [role, entries] of groups.entries()) {
            for (let offset = 0; offset < entries.length; offset += 100) {
                const batch = entries.slice(offset, offset + 100);
                const payload = {
                    action: "uploadRecords",
                    timerRole: role,
                    records: batch.map(entry => ({
                        recordId: entry.recordId,
                        eventId: entry.eventId || state.eventId,
                        timerRole: entry.timerRole,
                        bibNumber: String(entry.bibNumber),
                        distance: entry.distance,
                        finishTimestamp: entry.finishTimestamp,
                        elapsedTime: entry.elapsedTime || "",
                        entryTimestamp: entry.entryTimestamp || "",
                        deviceSessionId: entry.deviceSessionId || state.deviceSessionId,
                        notes: entry.notes || ""
                    }))
                };
                let result;
                try {
                    result = await postToAppsScript(payload);
                } catch (error) {
                    failures.push(role + ": " + error.message);
                    continue;
                }
                const accepted = new Set(result.accepted || []);
                const duplicates = new Set(result.duplicates || []);
                const conflicts = new Set(result.conflicts || []);
                for (const entry of batch) {
                    const id = String(entry.recordId);
                    if (conflicts.has(id)) {
                        failures.push(role + ": record ID conflict for " + id);
                        continue;
                    }
                    if (accepted.has(id) || duplicates.has(id)) {
                        entry.syncStatus = "synced";
                        entry.syncedAt = nowIso();
                        await saveUpdatedEntry(entry);
                        confirmedCount++;
                    } else {
                        failures.push(role + ": server did not confirm record " + id);
                    }
                }
                if (result.success !== true && !(result.conflicts || []).length) {
                    failures.push(role + ": " + (result.error || result.message || "server reported incomplete success."));
                }
            }
        }

        renderRecentEntries();
        updateCounters();
        if (failures.length) {
            setConnectionStatus("SYNC ISSUE");
            setSyncMessage(confirmedCount + " confirmed; " + failures.length + " issue(s). Unconfirmed records remain queued.");
            alert("Upload partially completed.\n\nConfirmed: " + confirmedCount + "\nIssues: " + failures.length + "\n\n" + failures.slice(0, 8).join("\n") + (failures.length > 8 ? "\nMore issues omitted." : "") + "\n\nUnconfirmed records remain saved locally.");
        } else {
            setConnectionStatus("CONNECTED");
            setSyncMessage("Upload confirmed for " + confirmedCount + " record(s).");
            alert("Upload complete. Confirmed records: " + confirmedCount);
        }
    } catch (error) {
        console.error("Upload failed:", error);
        setConnectionStatus(navigator.onLine ? "SYNC ISSUE" : "OFFLINE");
        setSyncMessage("Upload interrupted. Unconfirmed records remain queued. " + error.message);
        alert("Upload interrupted.\n\n" + error.message + "\n\nUnconfirmed records remain saved locally.");
    } finally {
        submitting = false;
        $("submitButton").disabled = false;
        renderRecentEntries();
        updateCounters();
    }
}

function getAvailableCategories() {
    const values = new Set();
    state.runners.forEach(runner => {
        const value = String(runner.category || "").trim();
        if (value) values.add(value);
    });
    if (serverReviewData && Array.isArray(serverReviewData.categories)) {
        serverReviewData.categories.forEach(value => {
            value = String(value || "").trim();
            if (value) values.add(value);
        });
    }
    state.entries.forEach(entry => {
        const runner = findRunner(entry.bibNumber);
        const value = String((runner && runner.category) || entry.category || "").trim();
        if (value) values.add(value);
    });
    return [...values].sort((a, b) => a.localeCompare(b));
}

function populateCategories() {
    const container = $("resultsCategoryTabs");
    if (!container) return;
    const categories = getAvailableCategories();
    const allowed = ["ALL", ...categories];
    if (!allowed.includes(selectedResultsCategory)) selectedResultsCategory = "ALL";

    container.innerHTML = "";
    const items = [{ value: "ALL", label: "All Categories" }, ...categories.map(c => ({ value: c, label: c }))];
    items.forEach(item => {
        const button = document.createElement("button");
        button.type = "button";
        button.className = "filter-tab" + (selectedResultsCategory === item.value ? " active" : "");
        button.dataset.category = item.value;
        button.setAttribute("aria-pressed", selectedResultsCategory === item.value ? "true" : "false");
        button.textContent = item.label;
        container.appendChild(button);
    });
}

function parseElapsedForResults(value) {
    const text = String(value || "").trim();
    const match = text.match(/^(\d{1,3}):(\d{2}):(\d{2})(?:\.(\d{1,3}))?$/);
    if (!match) return null;
    const fraction = Number((match[4] || "0").padEnd(3, "0"));
    return ((Number(match[1]) * 3600 + Number(match[2]) * 60 + Number(match[3])) * 1000) + fraction;
}

async function refreshResultsFromSheets(options = {}) {
    if (resultsRefreshing) return;
    if (!navigator.onLine) {
        resultsRefreshError = "Offline. Showing the last fetched server data or local records.";
        renderResults();
        return;
    }

    resultsRefreshing = true;
    resultsRefreshError = "";
    const button = $("refreshResultsButton");
    const refreshed = $("resultsLastRefreshed");
    if (button) {
        button.disabled = true;
        button.textContent = "Refreshing…";
    }
    if (refreshed) refreshed.textContent = "Fetching latest Google Sheets data…";

    try {
        const eventId = String(state.eventId || "").trim();
        const url = API_URL + "?action=reviewData" +
            (eventId ? "&eventId=" + encodeURIComponent(eventId) : "") +
            "&_=" + Date.now();
        const response = await fetch(url, {
            method: "GET",
            cache: "no-store",
            redirect: "follow"
        });
        if (!response.ok) throw new Error("HTTP " + response.status);
        const data = await response.json();
        if (!data || data.success !== true || !Array.isArray(data.records) || !Array.isArray(data.consolidated)) {
            throw new Error((data && data.error) || "Unexpected review-data response.");
        }
        serverReviewData = data;
        resultsRefreshError = "";
        populateCategories();
        renderSourceTabs();
        const stamp = new Date(data.fetchedAt || Date.now());
        if (refreshed) refreshed.textContent = "Last refreshed: " + stamp.toLocaleString();
        if (button) button.textContent = "↻ Refresh from Sheets";
        setConnectionStatus("CONNECTED");
        renderResults();
    } catch (error) {
        console.error("Results refresh failed:", error);
        resultsRefreshError = "Could not refresh from Google Sheets: " + error.message;
        if (refreshed) refreshed.textContent = "Refresh failed · previous data retained";
        if (button) button.textContent = "↻ Retry refresh";
        setConnectionStatus(navigator.onLine ? "SYNC ISSUE" : "OFFLINE");
        renderResults();
        if (!options.quiet) alert(resultsRefreshError + "\n\nExisting local records were not deleted.");
    } finally {
        resultsRefreshing = false;
        if (button) button.disabled = false;
    }
}

function renderResults() {
    const distance = $("resultsDistance").value;
    const category = selectedResultsCategory;
    const gender = $("resultsGender").value;
    const source = selectedResultsSource;
    let results = [];
    let usingServer = Boolean(serverReviewData);

    if (source === "CONSOLIDATED") {
        if (serverReviewData) {
            results = serverReviewData.consolidated.map(item => ({
                recordId: item.selectedRecordId || "",
                eventId: item.eventId || "",
                bibNumber: String(item.bibNumber || ""),
                runnerName: item.name || "Unregistered",
                category: item.category || "Unclassified",
                gender: item.gender || "Unknown",
                distance: item.distance || "",
                timerRole: item.selectedTimer || "—",
                elapsedTime: item.elapsedTime || "Pending",
                elapsedMs: parseElapsedForResults(item.elapsedTime),
                recordStatus: item.selectionStatus || "PROVISIONAL",
                reviewRequired: item.selectionStatus !== "FINAL",
                notes: item.notes || ""
            }));
        }
    } else if (serverReviewData) {
        results = serverReviewData.records
            .filter(entry => source === "ALL" || entry.timerRole === source)
            .map(entry => ({
                ...entry,
                runnerName: entry.runnerName || "Unregistered",
                elapsedMs: entry.elapsedMs === null || entry.elapsedMs === undefined
                    ? parseElapsedForResults(entry.elapsedTime)
                    : Number(entry.elapsedMs),
                recordStatus: entry.status || entry.recordStatus || "",
                reviewRequired: Boolean(entry.issues && entry.issues.length)
            }));
    } else {
        // Offline/initial fallback: show this device's local observations only.
        results = state.entries.filter(entry => source === "ALL" || entry.timerRole === source);
        usingServer = false;
    }

    results = results.filter(entry => {
        if (entry.distance !== distance) return false;
        if (entry.elapsedMs === null || !Number.isFinite(entry.elapsedMs) || entry.elapsedMs < 0) return false;
        if (source === "CONSOLIDATED" && !entry.recordId) return false;
        const runner = findRunner(entry.bibNumber);
        const actualCategory = String((runner && runner.category) || entry.category || "").trim();
        const actualGender = String((runner && runner.gender) || entry.gender || "").trim();
        if (category !== "ALL" && actualCategory !== category) return false;
        if (gender !== "ALL" && actualGender.toLowerCase() !== gender.toLowerCase()) return false;
        if (!usingServer && entry.recordStatus && entry.recordStatus !== "normal") return false;
        return true;
    });

    results.sort((a, b) => a.elapsedMs - b.elapsedMs || String(a.bibNumber).localeCompare(String(b.bibNumber), undefined, { numeric: true }));
    const wrap = $("resultsTableWrap");
    wrap.innerHTML = "";

    if (!results.length) {
        const message = source === "CONSOLIDATED"
            ? (serverReviewData ? "No consolidated results have been selected for this distance yet. Open Validation & Correction to select a candidate." : "Consolidated results have not been fetched yet. Click Refresh from Sheets.")
            : (usingServer ? "No server records match these filters." : "No local records match these filters. Refresh from Sheets to fetch observations from all timer devices.");
        wrap.innerHTML = '<div class="empty-state">' + message + '</div>';
    } else {
        const table = document.createElement("table");
        table.className = "results-table";
        const thead = document.createElement("thead");
        thead.innerHTML = "<tr><th>Rank</th><th>Bib</th><th>Name</th><th>Category</th><th>Gender</th><th>Time</th><th>Timer / Status</th></tr>";
        const tbody = document.createElement("tbody");
        results.forEach((entry, index) => {
            const row = document.createElement("tr");
            const runner = findRunner(entry.bibNumber);
            const timerLabel = source === "CONSOLIDATED"
                ? (entry.timerRole + " · " + (entry.recordStatus || "PROVISIONAL"))
                : (entry.timerRole + (entry.reviewRequired ? " · REVIEW" : (entry.recordStatus ? " · " + entry.recordStatus : "")));
            const values = [
                String(index + 1), entry.bibNumber,
                entry.runnerName || (runner && runner.name) || "Unregistered",
                entry.category || (runner && runner.category) || "Unclassified",
                entry.gender || (runner && runner.gender) || "Unknown",
                (Number.isFinite(entry.elapsedMs) && entry.elapsedMs >= 0 ? formatElapsed(entry.elapsedMs) : (entry.elapsedTime || "Pending")), timerLabel
            ];
            values.forEach((value, col) => {
                const cell = document.createElement("td");
                cell.textContent = value;
                if (col === 0) cell.className = "rank-number";
                row.appendChild(cell);
            });
            tbody.appendChild(row);
        });
        table.append(thead, tbody);
        wrap.appendChild(table);
    }

    const sourceLabel = source === "CONSOLIDATED" ? "consolidated selected result(s)" :
        (source === "ALL" ? "raw observation(s) across all timer sheets" : "raw observation(s) from " + source);
    let message = results.length + " " + sourceLabel + ". " +
        (source === "CONSOLIDATED" ? "Only race-director-selected results are shown." : "Multiple observations for the same bib are expected when timers are compared.");
    if (!usingServer) message += " Local-only fallback; click Refresh from Sheets to fetch other devices' uploaded records.";
    if (resultsRefreshError) message += " " + resultsRefreshError;
    if (results.some(entry => entry.reviewRequired)) message += " Some observations are flagged for review.";
    $("resultsMessage").textContent = message;
}

function exportCsv() {
    const columns = [
        "recordId",
        "eventId",
        "timerRole",
        "deviceSessionId",
        "bibNumber",
        "distance",
        "runnerName",
        "category",
        "gender",
        "registrationStatus",
        "finishTimestamp",
        "startTimestamp",
        "elapsedTime",
        "entryTimestamp",
        "recordStatus",
        "reviewRequired",
        "syncStatus",
        "notes"
    ];

    const escapeCsv = value =>
        '"' + String(value ?? "").replace(/"/g, '""') + '"';

    const lines = [
        columns.join(","),
        ...state.entries.map(entry =>
            columns.map(column => escapeCsv(entry[column])).join(",")
        )
    ];

    const blob = new Blob(["\uFEFF" + lines.join("\r\n")], {
        type: "text/csv;charset=utf-8"
    });

    const url = URL.createObjectURL(blob);
    const link = document.createElement("a");

    link.href = url;
    link.download =
        "velocity5000-" + state.timerRole + "-backup.csv";

    link.click();
    setTimeout(() => URL.revokeObjectURL(url), 1000);
}

        const key = button.dataset.key;

        if (key === "backspace") {
            backspaceBib();
        } else if (key === "enter") {
            submitBib();
        } else if (/^\d$/.test(key)) {
            appendBibDigit(key);

    });

    // Keep native mobile keyboard out of the way.
    // Physical keyboard input is handled explicitly on desktop.

        renderBibLookup();

    // Laptop keyboard support.

        const active = document.activeElement;
        const isBibField = active === $("bibInput");

        if (active && active.matches("input, textarea, select") && !isBibField) {
            return;
        }

        if (/^\d$/.test(event.key)) {
            event.preventDefault();
            appendBibDigit(event.key);
        } else if (event.key === "Backspace") {
            event.preventDefault();
            backspaceBib();
        } else if (event.key === "Enter") {
            event.preventDefault();
            submitBib();
        }

        showScreen("results");

        showScreen("timing");

        showScreen("timing");

    `;

    }
}

function formatElapsed(value) {
  if (value === null || value === undefined || value === "") return "—";
  const raw = String(value).trim();
  const match = raw.match(/^(?:(\d+):)?([0-5]?\d):([0-5]?\d)(?:\.(\d+))?$/);
  if (match) {
    const hours = Number(match[1] || 0), minutes = Number(match[2] || 0), seconds = Number(match[3] || 0);
    const hundredths = Number(((match[4] || "0") + "00").slice(0, 2));
    return String(hours).padStart(2, "0") + ":" + String(minutes).padStart(2, "0") + ":" + String(seconds).padStart(2, "0") + "." + String(hundredths).padStart(2, "0");
  }
  return raw;
}
function formatFinishClock(value) {
  if (!value || isNaN(Date.parse(value))) return "—";
  const date = new Date(value);
  try {
    const parts = new Intl.DateTimeFormat("en-GB", {
      timeZone: "Asia/Manila", hour: "2-digit", minute: "2-digit", second: "2-digit",
      hourCycle: "h23", fractionalSecondDigits: 2
    }).formatToParts(date);
    const p = Object.fromEntries(parts.map(part => [part.type, part.value]));
    return `${p.hour || "00"}:${p.minute || "00"}:${p.second || "00"}.${p.fractionalSecond || "00"}`;
  } catch (_) {
    return new Intl.DateTimeFormat("en-GB", { timeZone: "Asia/Manila", hour: "2-digit", minute: "2-digit", second: "2-digit", hour12: false }).format(date) + "." + String(Math.floor(date.getMilliseconds() / 10)).padStart(2, "0");
  }
}

async function getReviewData() {
  if (loading) return;
  loading = true;
  const button = $("refreshButton");
  button.disabled = true;
  button.textContent = "Refreshing…";
  setMessage("Fetching latest timer sheets…");
  try {
    // Fetch all events; event/category/distance filters are applied locally so they remain stable.
    const url = API_URL + "?action=reviewData&_=" + Date.now();
    const response = await fetch(url, { method: "GET", cache: "no-store", redirect: "follow" });
    if (!response.ok) throw new Error("HTTP " + response.status);
    const data = await response.json();
    if (!data || data.success !== true || !Array.isArray(data.records) || !Array.isArray(data.consolidated)) {
      throw new Error((data && data.error) || "Unexpected review-data response.");
    }
    reviewData = data;
    $("connectionStatus").textContent = "CONNECTED";
    populateFilters();
    renderAll();
    const refreshed = new Date(data.fetchedAt || Date.now()).toLocaleString();
    setMessage("Loaded " + data.records.length + " observations · Refreshed " + refreshed + ".");
  } catch (error) {

    $("connectionStatus").textContent = navigator.onLine ? "SYNC ISSUE" : "OFFLINE";

function buildGroups() {
  const groups = new Map();
  (reviewData.records || []).forEach(record => {
    const key = keyFor(record.eventId, record.distance, record.bibNumber);
    if (!groups.has(key)) groups.set(key, { key, eventId: record.eventId, distance: record.distance, bibNumber: record.bibNumber, records: [] });
    groups.get(key).records.push(record);
  });
  const consolidated = new Map();
  (reviewData.consolidated || []).forEach(item => consolidated.set(keyFor(item.eventId, item.distance, item.bibNumber), item));
  groups.forEach(group => {
    group.records.sort((a,b) => {
      const rank = (a.timerRole === "Timer1" ? 0 : 1) - (b.timerRole === "Timer1" ? 0 : 1);
      if (rank) return rank;
      const at = Number.isFinite(a.elapsedMs) ? a.elapsedMs : Infinity;
      const bt = Number.isFinite(b.elapsedMs) ? b.elapsedMs : Infinity;
      return at - bt;
    });
    group.activeRecords = group.records.filter(r => !r.excluded);
    group.selected = consolidated.get(group.key) || null;
    group.selectedRecord = group.selected && group.selected.selectedRecordId
      ? group.records.find(r => r.recordId === group.selected.selectedRecordId) || null : null;
    group.status = getGroupStatus(group);
  });

function getGroupStatus(group) {
  if (!group.activeRecords.length) return "EXCLUDED";
  if (group.activeRecords.some(r => (r.issues || []).includes("INVALID_BIB") || (r.issues || []).includes("INVALID_BIB_FORMAT"))) return "INVALID BIB";
  if (group.activeRecords.some(r => (r.issues || []).includes("DISTANCE_MISMATCH") || (r.issues || []).includes("INVALID_OR_MISSING_ELAPSED_TIME") || (r.issues || []).includes("INVALID_FINISH_TIMESTAMP"))) return "NEEDS REVIEW";
  if (group.selected && group.selected.selectionStatus === "FINAL") return "FINAL";
  const selectedId = group.selected && String(group.selected.selectedRecordId || "");
  const selectedBy = group.selected && String(group.selected.updatedBy || "");
  const humanConfirmed = Boolean(selectedId) && !/^AUTO:/i.test(selectedBy);
  if (humanConfirmed && group.selected.selectedTimer !== "Timer1") return "SELECTED BACKUP";
  if (humanConfirmed) return "SELECTED";
  if (group.activeRecords.some(r => (r.issues || []).includes("DUPLICATE_CANDIDATE"))) return "DUPLICATE CANDIDATE";
  if (!group.activeRecords.some(r => r.timerRole === "Timer1")) return "MISSING PRIMARY";
  if (group.selected && group.selected.selectedRecordId) return "SELECTED";
  return "NEEDS REVIEW";
}

  const issueRecords = group.activeRecords.filter(r => (r.issues || []).length);
  if (status === "ISSUES" && !issueRecords.length && !["DUPLICATE CANDIDATE", "MISSING PRIMARY", "INVALID BIB", "NEEDS REVIEW"].includes(group.status)) return false;

      '<div><div class="candidate-title">' + escapeHtml(record.timerRole) + ' · ' + escapeHtml(formatElapsed(record.elapsedTime)) + (selected ? ' <span class="issue-pill valid">CURRENT</span>' : "") + '</div>' +
      '<div class="candidate-details">Status: ' + escapeHtml(record.status || "VALID") + '<br>Finish: ' + escapeHtml(formatFinishClock(record.finishTimestamp)) + raw + '</div></div>' +
      '<div class="candidate-actions">' + selectBtn + toggleBtn + correctBtn + '</div></article>';
  }).join("");
  const dialog = $("candidateDialog");
  if (typeof dialog.showModal === "function") dialog.showModal(); else dialog.setAttribute("open", "open");
}
function closeDialog() {
  const dialog = $("candidateDialog");
  if (typeof dialog.close === "function") dialog.close(); else dialog.removeAttribute("open");
  activeDialogKey = null; pendingCandidateId = "";
}
async function postAction(payload) {
  if (!canEdit) throw new Error("Only Timer1 can change results in this test build.");
  const response = await fetch(API_URL, {
    method: "POST", redirect: "follow",
    headers: { "Content-Type": "text/plain;charset=utf-8" },
    body: JSON.stringify(actorPayload(payload))
  });
  if (!response.ok) throw new Error("HTTP " + response.status);
  const result = await response.json();
  if (!result || result.success !== true) throw new Error((result && result.error) || "Action failed.");
  return result;
}
async function runAction(payload, successMessage) {
  try {
    setMessage("Saving…");
    const result = await postAction(payload);
    if (payload.action === "excludeRecord" || payload.action === "excludeRecords") {
      // Exclusions become visible immediately after the backend confirms and data refreshes.
      selectedGroups.clear();
    }
    if (successMessage) setMessage(successMessage);
    await getReviewData();
    if (result.message) setMessage(result.message);
    return result;
  } catch (error) {
    setMessage("Could not save: " + error.message, true);
    alert("Could not save: " + error.message);
    return null;
  }
}
async function confirmCandidateSelection() {
  if (!pendingCandidateId || !activeDialogKey) return;
  const group = groupsByKey.get(activeDialogKey);
  const candidate = group && group.records.find(r => r.recordId === pendingCandidateId);
  if (!candidate) return;
  if (!confirm("Use " + candidate.timerRole + " time " + formatElapsed(candidate.elapsedTime) + " for bib " + candidate.bibNumber + "?")) return;
  const result = await runAction({ action: "selectResult", recordId: candidate.recordId, eventId: candidate.eventId }, "Selection saved.");
  if (result) closeDialog();
}
async function toggleObservation(action, recordId) {
  const record = reviewData.records.find(r => r.recordId === recordId);
  if (!record) return;
  const label = action === "excludeRecord" ? "exclude this observation" : "restore this observation";
  if (!confirm("Confirm: " + label + " for bib " + record.bibNumber + " (" + record.timerRole + ")?")) return;
  const result = await runAction({ action, recordId: recordId, eventId: record.eventId }, "Updated.");
  if (result && activeDialogKey) openGroupDialog(activeDialogKey);
}
async function correctBibForRecords(recordIds, currentBib) {
  if (!canEdit || !recordIds.length) return;
  const newBib = prompt("Enter the correct registered bib number:", currentBib || "");
  if (newBib === null) return;
  const cleanBib = newBib.trim();
  if (!/^\d+$/.test(cleanBib)) { alert("Enter a numeric bib number."); return; }
  if (!confirm("Correct " + recordIds.length + " observation(s) to bib " + cleanBib + "?")) return;
  const result = await runAction({ action: "correctBib", recordIds: recordIds, newBibNumber: cleanBib }, "Bib correction saved.");
  if (result && activeDialogKey) closeDialog();
}
async function correctGroupBib(key) {
  const group = groupsByKey.get(key);
  if (!group) return;
  const targets = group.activeRecords.filter(r => (r.issues || []).includes("INVALID_BIB") || (r.issues || []).includes("INVALID_BIB_FORMAT"));
  if (!targets.length) { alert("This bib group has no active invalid-bib observations."); return; }
  await correctBibForRecords(targets.map(r => r.recordId), group.bibNumber);
}
async function handleGroupAction(action, key) {
  if (action === "review") { openGroupDialog(key); return; }
  if (action === "correct-group") { await correctGroupBib(key); return; }
}
async function bulkExcludeRestore(action) {
  const keys = selectedGroupKeys();
  if (!keys.length) { alert("Select one or more bib rows first."); return; }
  const records = keys.flatMap(key => (groupsByKey.get(key)?.records || []));
  const ids = [...new Set(records.filter(r => action === "excludeRecords" ? !r.excluded : r.excluded).map(r => r.recordId))];
  if (!ids.length) { alert(action === "excludeRecords" ? "The selected rows have no active observations." : "The selected rows have no excluded observations."); return; }
  const verb = action === "excludeRecords" ? "exclude all active observations for" : "restore excluded observations for";
  if (!confirm("Confirm: " + verb + " " + keys.length + " selected bib result(s)?")) return;
  await runAction({ action, recordIds: ids }, "Updated " + ids.length + " observation(s).");
}
async function bulkTimer1Defaults() {
  const keys = selectedGroupKeys();
  if (!keys.length) { alert("Select one or more bib rows first."); return; }
  const selectedIds = [], skipped = [];
  keys.forEach(key => {
    const group = groupsByKey.get(key);
    if (!group) { skipped.push(key); return; }

    const activeTimer1 = group.activeRecords.filter(r => r.timerRole === "Timer1");
    let candidate = null;

    if (activeTimer1.length === 1 && isSelectable(activeTimer1[0])) {
      candidate = activeTimer1[0];
    } else if (activeTimer1.length === 0) {
      const backups = group.activeRecords
        .filter(isSelectable)
        .filter(r => r.timerRole !== "Timer1")
        .sort((a, b) => a.elapsedMs - b.elapsedMs ||
          String(a.timerRole).localeCompare(String(b.timerRole)));
      candidate = backups[0] || null;
    }

    if (candidate) selectedIds.push(candidate.recordId);
    else skipped.push(group.bibNumber || key);
  });

  if (!selectedIds.length) {
    alert("No selected bibs have an unambiguous valid default. Bibs with duplicate Timer1 observations or invalid times need individual review.");
    return;
  }

  if (!confirm("Apply defaults to " + selectedIds.length + " selected bib(s)? Timer1 is preferred when one valid observation exists. If Timer1 has no observation, the fastest valid backup is suggested. Review and finalize remain required.")) return;

  const result = await runAction(
    { action: "selectResults", recordIds: selectedIds, applyDefaults: true },
    "Defaults applied to selected bibs."
  );
  if (result && skipped.length) {
    alert("Updated " + selectedIds.length + " result(s). These bibs need individual review: " + skipped.join(", "));
  }
  selectedGroups.clear();
  renderGroups();
}
async function finalizeEvent() {
  const eventId = $("eventFilter").value;
  if (!eventId) { alert("Select one event before finalizing."); return; }
  if (!confirm("Finalize the currently selected Consolidated results for " + eventId + "? This can be amended later, but results will be marked FINAL.")) return;
  await runAction({ action: "finalizeResults", eventId: eventId }, "Event finalized.");
}

const CACHE_NAME = "v5000-shell-v5";
const SHELL = ["./", "./index.html", "./style.css", "./style-additions.css", "./app.js", "./review.html", "./review.css", "./review.js", "./manifest.json"];
self.addEventListener("install", event => {
  event.waitUntil(caches.open(CACHE_NAME).then(cache => cache.addAll(SHELL)).then(() => self.skipWaiting()));
});
self.addEventListener("activate", event => {
  event.waitUntil(caches.keys().then(keys => Promise.all(keys.filter(key => key !== CACHE_NAME).map(key => caches.delete(key)))).then(() => self.clients.claim()));
});
self.addEventListener("fetch", event => {
  const req = event.request;
  const url = new URL(req.url);
  if (req.method !== "GET" || url.origin !== self.location.origin) return;
  if (url.pathname.includes("/macros/")) return;
  event.respondWith(fetch(req).then(response => {
    if (response && response.ok) {
      const copy = response.clone();
      caches.open(CACHE_NAME).then(cache => cache.put(req, copy));
    }
    return response;
  }).catch(() => caches.match(req).then(cached => cached || caches.match("./index.html"))));
});
